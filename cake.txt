const int MAXN = 2e5 + 5;
ll tree[4 * MAXN];
ll lazy[4 * MAXN];
int a[MAXN];

// Push the lazy value from parent to children
void push(int v, int tl, int tr) {
if (lazy[v] != 0) {
int tm = (tl + tr) / 2;
// Apply pending update to children's sums
tree[2 * v] += lazy[v] * (tm - tl + 1);
tree[2 * v + 1] += lazy[v] * (tr - tm);
// Pass the lazy tag to children
lazy[2 * v] += lazy[v];
lazy[2 * v + 1] += lazy[v];
// Reset current lazy tag
lazy[v] = 0;
}
}

void build(int v, int tl, int tr) {
if (tl == tr) {
tree[v] = a[tl];
} else {
int tm = (tl + tr) / 2;
build(2 * v, tl, tm);
build(2 * v + 1, tm + 1, tr);
tree[v] = tree[2 * v] + tree[2 * v + 1];
}
}

void update(int v, int tl, int tr, int l, int r, int add) {
if (l > r) return;
if (l == tl && r == tr) {
tree[v] += (ll)add * (tr - tl + 1);
lazy[v] += add;
} else {
push(v, tl, tr);
int tm = (tl + tr) / 2;
update(2 * v, tl, tm, l, min(r, tm), add);
update(2 * v + 1, tm + 1, tr, max(l, tm + 1), r, add);
tree[v] = tree[2 * v] + tree[2 * v + 1];
}
}

struct MinHeap {
vector h;

// Sift down an element to maintain heap property
void shiftDown(int i) {
int n = h.size();
while (true) {
int smallest = i;
int l = 2 * i + 1, r = 2 * i + 2;

if (l < n && h[l] < h[smallest]) smallest = l;
if (r < n && h[r] < h[smallest]) smallest = r;

if (smallest == i) break;
swap(h[i], h[smallest]);
i = smallest;
}
}

// Sift up a new element to maintain heap property
void shiftUp(int i) {
while (i > 0 && h[i] < h[(i - 1) / 2]) {
swap(h[i], h[(i - 1) / 2]);
i = (i - 1) / 2;
}
}

void push(int x) {
h.push_back(x);
shiftUp(h.size() - 1);
}

void pop() {
if (h.empty()) return;
h[0] = h.back();
h.pop_back();
if (!h.empty()) shiftDown(0);
}

bool empty() { return h.empty(); }

Node *min_node = NULL;
int total_nodes = 0;

// Helper: Add node to a circular doubly linked list
void addToRootList(Node* node) {
if (!min_node) {
min_node = node;
} else {
node->left = min_node;
node->right = min_node->right;
min_node->right->left = node;
min_node->right = node;
if (node->key < min_node->key) min_node = node;
}
}

void insert(int key) {
Node* node = new Node(key);
addToRootList(node);
total_nodes++;
}

// Helper: Link y as child of x
void link(Node* y, Node* x) {
// Remove y from root list
y->left->right = y->right;
y->right->left = y->left;
// Make y child of x
y->parent = x;
if (!x->child) {
x->child = y;
y->left = y->right = y;
} else {
y->left = x->child;
y->right = x->child->right;
x->child->right->left = y;
x->child->right = y;
}
x->degree++;
y->mark = false;
}

void consolidate() {
int max_deg = log2(total_nodes) + 2;
vector A(max_deg, NULL);

// Get all roots in a vector first because the list changes
vector roots;
Node* curr = min_node;
if (curr) {
do { roots.push_back(curr); curr = curr->right; } while (curr != min_node);
}

for (Node* x : roots) {
int d = x->degree;
while (A[d]) {
Node* y = A[d];
if (x->key > y->key) swap(x, y);
link(y, x);
A[d] = NULL;
d++;
}
A[d] = x;
}

min_node = NULL;
for (Node* node : A) {
if (node) {
node->left = node->right = node; // Reset pointers for new list
addToRootList(node);
}
}
}

int extractMin() {
Node* z = min_node;
if (!z) return -1;

if (z->child) {
Node* child = z->child;
do {
Node* next = child->right;
addToRootList(child);
child->parent = NULL;
child = next;
} while (child != z->child);
}

z->left->right = z->right;
z->right->left = z->left;

int res = z->key;
if (z == z->right) min_node = NULL;
else {
min_node = z->right;
consolidate();
}
total_nodes--;
return res;
}

void cut(Node* x, Node* y) {
// Remove x from child list of y
if (x == x->right) y->child = NULL;
else {
x->left->right = x->right;
x->right->left = x->left;
if (y->child == x) y->child = x->right;
}
y->degree--;
addToRootList(x);
x->parent = NULL;
x->mark = false;
}

void cascadingCut(Node* y) {
Node* z = y->parent;
if (z) {
if (!y->mark) y->mark = true;
else {
cut(y, z);
cascadingCut(z);
}
}
}

void decreaseKey(Node* x, int k) {
if (k > x->key) return;
x->key = k;
Node* y = x->parent;
if (y && x->key < y->key) {
cut(x, y);
cascadingCutYes;
}
if (x->key < min_node->key) min_node = x;
}

// Find node for delete/decrease (O(N) search)
Node* findNode(Node* n, int k) {
if (!n) return NULL;
Node* temp = n;
do {
if (temp->key == k) return temp;
Node* res = findNode(temp->child, k);
if (res) return res;
temp = temp->right;
} while (temp != n);
return NULL;
}
